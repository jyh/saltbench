/-- Summing a list of quotients with a fixed denominator. -/
lemma iso_sum_map_div (l : List Rat) (f : Rat → Rat) (c : Rat) :
    (l.map (fun x => f x / c)).sum = (l.map f).sum / c := by
  induction l with
  | nil => simp
  | cons a t ih => simp [ih, add_div]

/-- Each term of a sum of absolute values is nonnegative, hence so is the sum. -/
lemma iso_sum_abs_nonneg (l : List Rat) (g : Rat → Rat) :
    0 ≤ (l.map (fun x => |g x|)).sum := by
  apply List.sum_nonneg
  intro x hx
  simp only [List.mem_map] at hx
  obtain ⟨y, _, rfl⟩ := hx
  exact abs_nonneg _

/-- The two ways of writing the mean absolute deviation agree up to one factor of
`numbers.length`: `(Σ |x - mean|) * n = Σ |x * n - Σ x|`. -/
lemma iso_mad_sum (numbers : List Rat) (h : numbers ≠ []) :
    (numbers.map (fun x => |x - numbers.sum / (numbers.length : Rat)|)).sum
        * (numbers.length : Rat)
      = (numbers.map (fun x => |x * (numbers.length : Rat) - numbers.sum|)).sum := by
  have hpos : (0 : Rat) < (numbers.length : Rat) := by
    have : 0 < numbers.length := List.length_pos_of_ne_nil h
    exact_mod_cast this
  have hne : (numbers.length : Rat) ≠ 0 := ne_of_gt hpos
  have hfun : (fun x : Rat => |x - numbers.sum / (numbers.length : Rat)|)
      = (fun x : Rat => |x * (numbers.length : Rat) - numbers.sum| / (numbers.length : Rat)) := by
    funext x
    have hx : x * (numbers.length : Rat) - numbers.sum
        = (x - numbers.sum / (numbers.length : Rat)) * (numbers.length : Rat) := by
      field_simp
    rw [hx, abs_mul, abs_of_pos hpos, mul_div_assoc, div_self hne, mul_one]
  rw [hfun, iso_sum_map_div, div_mul_cancel₀ _ hne]

/--

lemma iso_forward_direction_false :
    ¬ (∀ impl : List Rat → Rat,
        (∀ numbers, problem_spec impl numbers) → ∀ numbers, generated_spec impl numbers) := by
