== AGENT iso helper lemmas ==
set_option linter.deprecated false

/-- `String.data` is the deprecated spelling of `String.toList`. -/
theorem iso_data_eq_toList (s : String) : s.data = s.toList := rfl

/-- The characters of a string obtained by joining single-character strings. -/
theorem iso_join_map_singleton (l : List Char) :
    (String.join (l.map (fun c => String.mk [c]))).toList = l := by
  have hmk : ∀ c : Char, (String.mk [c]).toList = [c] := fun c => String.toList_ofList
  simp only [String.toList_join, List.map_map, Function.comp_def, hmk]
  induction l with
  | nil => simp
  | cons a t ih => simp [ih]

/-- Dropping the first character of a string drops the first character of its list. -/
theorem iso_toList_drop_one (s : String) :
    ((s.drop 1).toString).toList = s.toList.drop 1 := by
  by_cases hs : s = ""
  · subst hs; rfl
  · have h : s.startPos ≠ s.endPos := by
      simpa [String.startPos_eq_endPos_iff] using hs
    have hdrop : s.drop 1 = s.sliceFrom (s.startPos.next h) := by
      simp [String.drop, String.Slice.drop, String.Slice.Pos.nextn, String.sliceFrom,
        String.startPos_toSlice, String.Pos.toSlice_next, String.Pos.toSlice_inj, h]
    have hempty : (s.sliceTo s.startPos).copy = "" := by
      apply String.utf8ByteSize_eq_zero_iff.mp
      simp [String.Slice.utf8ByteSize_copy, String.endExclusive_sliceTo,
        String.startInclusive_sliceTo, String.offset_startPos]
    have key := String.Pos.eq_copy_sliceTo_append_get h
    rw [hempty] at key
    have hlist := congrArg String.toList key
    rw [String.toList_append, String.toList_append, String.toList_empty,
      String.toList_singleton] at hlist
    rw [hdrop]
    show ((s.sliceFrom (s.startPos.next h)).copy).toList = s.toList.drop 1
    rw [hlist]
    simp

/-- Deleting one character and then the rest is the same as deleting all of them at once. -/
theorem iso_filter_step (a : Char) (rest l : List Char) :
    (l.filter (fun x => decide (x ≠ a))).filter (fun ch => !(rest.contains ch))
      = l.filter (fun ch => !((a :: rest).contains ch)) := by
  rw [List.filter_filter]
  apply List.filter_congr
  intro x _
  simp [Bool.and_comm]
