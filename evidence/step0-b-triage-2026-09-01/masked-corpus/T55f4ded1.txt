/-- Under `problem_spec` at every `n`, membership in `impl n` is exactly primality below `n`.
    This is the half of `generated_spec` that `problem_spec` really does pin down. -/
theorem mem_impl_iff_of_problem_spec {impl : Nat → List Nat}
    (h : ∀ n, problem_spec impl n) (n i : Nat) :
    i ∈ impl n ↔ (Nat.Prime i ∧ i < n) := by
  obtain ⟨r, hr, hs⟩ := h n
  subst hr
  cases n with
  | zero =>
    have h0 : impl 0 = [] := hs
    simp [h0]
  | succ m =>
    obtain ⟨hlt, hmem⟩ := hs (Nat.succ_pos m)
    constructor
    · intro hi
      obtain ⟨j, hj, hji⟩ := List.mem_iff_getElem.mp hi
      have hjj := hlt j hj
      rw [getElem!_pos (impl (m + 1)) j hj, hji] at hjj
      exact hjj
    · rintro ⟨hp, hi⟩
      exact hmem i hi hp

/-- The direction that is true: `generated_spec` implies `problem_spec`. -/
theorem problem_spec_of_generated_spec {impl : Nat → List Nat}
    (h : ∀ n, generated_spec impl n) (n : Nat) : problem_spec impl n := by
  refine ⟨impl n, rfl, ?_⟩
  cases n with
  | zero =>
    show impl 0 = []
    refine List.eq_nil_iff_forall_not_mem.mpr ?_
    intro a ha
    exact absurd (((h 0).1 a).mp ha).2 (Nat.not_lt_zero a)
  | succ m =>
    intro _
    refine ⟨?_, ?_⟩
    · intro i hi
      rw [getElem!_pos (impl (m + 1)) i hi]
      exact ((h (m + 1)).1 _).mp (List.getElem_mem hi)
    · intro i hi hp
      exact ((h (m + 1)).1 i).mpr ⟨hp, hi⟩

/-- A short restatement of what was actually landed, derived rather than asserted: an
    implementation satisfying `generated_spec` everywhere satisfies `problem_spec` everywhere,
    and consequently returns `[]` on `0` and always lists `2` once `n` exceeds `2`. -/
theorem generated_spec_sound {impl : Nat → List Nat} (h : ∀ n, generated_spec impl n) :
    (∀ n, problem_spec impl n) ∧ impl 0 = [] ∧ (∀ n, 2 < n → 2 ∈ impl n) := by
  refine ⟨problem_spec_of_generated_spec h, ?_, ?_⟩
  · obtain ⟨r, hr, hs⟩ := problem_spec_of_generated_spec h 0
    subst hr
    exact hs
  · intro n hn
    exact ((h n).1 2).mpr ⟨Nat.prime_two, hn⟩
