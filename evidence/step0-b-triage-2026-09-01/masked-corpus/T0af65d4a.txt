        simpa using String.get_of_valid ([] : List Char) [c]

        have := String.next_of_valid ([] : List Char) c []

theorem front?_ofList_cons (hd : Char) (tl : List Char) : (String.ofList (hd :: tl)).front? = some hd := by

theorem front_alpha_iff (s : String) (hs : s ≠ "") :
    s.front?.map Char.isAlpha = some true ↔ Char.isAlpha (s.get! (0 : String.Pos.Raw)) := by
  obtain ⟨l, rfl⟩ := s.exists_eq_ofList
  match l, hs with
  | (hd :: tl), _ =>
    have hget : (String.ofList (hd :: tl)).get! (0 : String.Pos.Raw) = hd := by
      show String.Pos.Raw.get! (String.ofList (hd :: tl)) 0 = hd
      rw [String.Pos.Raw.get!_eq_get]
      simpa using String.get_of_valid ([] : List Char) (hd :: tl)
    rw [front?_ofList_cons, hget]
    simp

theorem splitOnP_eq_singleton_iff {α : Type} (p : α → Bool) :
    ∀ (l w : List α), l.splitOnP p = [w] ↔ (l = w ∧ ∀ y ∈ w, ¬ p y)
  | [], w => by
      rw [List.splitOnP_nil]
      constructor
      · intro h
        have hw : ([] : List α) = w := (List.cons_eq_cons.mp h).1
        exact ⟨hw, by rw [← hw]; simp⟩
      · rintro ⟨hw, -⟩
        rw [← hw]
  | (x :: l'), w => by
      rw [List.splitOnP_cons]
      by_cases hpx : p x = true
      · simp only [hpx, if_true]
        constructor
        · intro h
          exact absurd (congrArg List.length h)
            (by simpa using List.splitOnP_ne_nil p l')
        · rintro ⟨h1, h2⟩
          exact absurd (h2 x (h1 ▸ List.mem_cons_self)) (by simp [hpx])
      · simp only [hpx, Bool.false_eq_true, if_false]
        obtain ⟨w0, rest, hm⟩ := List.exists_cons_of_ne_nil (List.splitOnP_ne_nil p l')
        rw [hm, List.modifyHead]
        constructor
        · intro h
          have h1 : x :: w0 = w := (List.cons_eq_cons.mp h).1
          have h2 : rest = [] := (List.cons_eq_cons.mp h).2
          have hw0 : l'.splitOnP p = [w0] := by rw [hm, h2]
          rw [splitOnP_eq_singleton_iff p l' w0] at hw0
          refine ⟨by rw [← h1, ← hw0.1], ?_⟩
          rintro y hy
          rw [← h1] at hy
          rcases List.mem_cons.mp hy with rfl | hy
          · simpa using hpx
          · exact hw0.2 y hy
        · rintro ⟨rfl, h2⟩
          have hw0 : l'.splitOnP p = [l'] :=
            (splitOnP_eq_singleton_iff p l' l').mpr
              ⟨rfl, fun y hy => h2 y (List.mem_cons_of_mem x hy)⟩
          rw [hw0] at hm
          have h2 : rest = [] := (List.cons_eq_cons.mp hm).2.symm
          have h1 : w0 = l' := (List.cons_eq_cons.mp hm).1.symm
          rw [h1, h2]

theorem splitOnP_eq_two_imp {α : Type} (p : α → Bool) :
    ∀ (l xs after : List α), l.splitOnP p = [xs, after] →
      (∀ y ∈ xs, ¬ p y) ∧ ∃ sep, p sep = true ∧ l = xs ++ sep :: after ∧ ∀ y ∈ after, ¬ p y
  | [], xs, after => by
      intro h
      rw [List.splitOnP_nil] at h
      exact absurd (congrArg List.length h) (by simp)
  | (x :: l'), xs, after => by
      intro h
      rw [List.splitOnP_cons] at h
      by_cases hpx : p x = true
      · simp only [hpx, if_true] at h
        have h1 : ([] : List α) = xs := (List.cons_eq_cons.mp h).1
        have h2 : l'.splitOnP p = [after] := (List.cons_eq_cons.mp h).2
        rw [splitOnP_eq_singleton_iff p l' after] at h2
        refine ⟨by rw [← h1]; simp, x, hpx, ?_, h2.2⟩
        rw [← h1, List.nil_append, h2.1]
      · simp only [hpx, Bool.false_eq_true, if_false] at h
        obtain ⟨w0, rest, hm⟩ := List.exists_cons_of_ne_nil (List.splitOnP_ne_nil p l')
        rw [hm, List.modifyHead] at h
        have h1 : x :: w0 = xs := (List.cons_eq_cons.mp h).1
        have h2 : rest = [after] := (List.cons_eq_cons.mp h).2
        have hrec : l'.splitOnP p = [w0, after] := by rw [hm, h2]
        obtain ⟨hw0, sep, hsep, hleq, hafter⟩ := splitOnP_eq_two_imp p l' w0 after hrec
        refine ⟨?_, sep, hsep, ?_, hafter⟩
        · rw [← h1]
          rintro y hy
          rcases List.mem_cons.mp hy with rfl | hy
          · simpa using hpx
          · exact hw0 y hy
        · rw [← h1, hleq]
